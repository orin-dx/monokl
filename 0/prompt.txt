```rust
let head = &src[..src.find("#[cfg(test)]").expect("no test module marker")];
fn strip_vis(line: &str) -> &str {
    let l = line.trim_start();
    match l.strip_prefix("pub") {
        Some(rest) => {
            let rest = if rest.starts_with('(') {
                match rest.find(')') { Some(i) => &rest[i + 1..], None => rest }
            } else { rest };
            rest.trim_start()
        }
        None => l,
    }
}
let decl = |kw: &str| -> Vec<&str> {
    head.lines().filter(|l| strip_vis(l).starts_with(kw)).map(str::trim).collect()
};
assert_eq!(decl("mod "), vec!["pub mod lexer;", "pub mod parser;", "pub mod plan;"]);
assert_eq!(decl("use "), vec!["pub use parser::{Modifier, ParsedTerm, parse};", "pub use plan::QueryPlan;"]);
```
